def initialize(analysis, owners: [], **details)
  @analysis = analysis
  @owners = Array(owners)
  @capability = Ast::Merge::Comment::Capability.native_full(
    source: :prism,
    style: :hash_comment,
    attachment_hints: true,
    comment_nodes: true,
    owner_count: @owners.size,
    comment_count: @analysis.tree.comments.size,
    **details
  )
  @attachments_by_owner = {}
  @preamble_region = nil
  @postlude_region = nil
  @orphan_regions = []
  build!
end
