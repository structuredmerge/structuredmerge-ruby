def initialize(type, start_byte:, end_byte:, source:, children: [], fields: {})
  @type = type.to_s
  @start_byte = start_byte
  @end_byte = end_byte
  @source = source
  @children = children
  @fields = fields
  @children.each { |child| child.parent = self }
  @fields.each_value { |child| child.parent = self }
end
