def initialize(source)
  @source = source.to_s
  @backend = BACKEND_REFERENCE.id.to_sym
  @errors = []
  parse!
rescue StandardError => e
  @root_node = nil
  @wrappers = [].freeze
  @document_semantics = nil
  @issues = [].freeze
  @errors = [e].freeze
end
