def initialize(source, dialect: :typescript)
  @source = source
  @dialect = dialect.to_sym
  @backend = TREE_SITTER_BACKEND.id.to_sym
  @errors = []
  parse!
rescue StandardError => e
  @root_node = nil
  @declarations = [].freeze
  @errors = [e].freeze
end
